```python
    return await fn(**args)
```

```python
import random

class SlippageModel:
    """Simulates realistic market slippage for paper trading."""

    def __init__(
        self,
        base_slippage_pct: float = 0.05,    # 0.05% base slippage
        max_slippage_pct: float = 0.3,       # 0.3% max for illiquid stocks
        impact_factor: float = 0.0001,        # Market impact per ₹1L of order value
    ):
        self.base_slippage_pct = base_slippage_pct
        self.max_slippage_pct = max_slippage_pct
        self.impact_factor = impact_factor

    def calculate_slippage(self, price: float, quantity: int, action: str) -> float:
        """Calculate slippage-adjusted fill price."""
        order_value = price * quantity

        # Base slippage (random component)
        base = self.base_slippage_pct / 100.0
        random_slippage = random.uniform(0, base)

        # Market impact component (larger orders = more impact)
        impact = self.impact_factor * (order_value / 100000)  # per ₹1L

        # Total slippage percentage
        total_slippage = min(random_slippage + impact, self.max_slippage_pct / 100.0)

        # Apply direction: BUY pays more, SELL receives less
        if action == "BUY":
            return price * (1 + total_slippage)
        else:
            return price * (1 - total_slippage)

# Add to PaperTradingEngine.__init__:
def __init__(self, initial_cash: float = 1000000.0, slippage_model: SlippageModel | None = None):
```

```python
def _generate_embedding(text: str) -> list[float]:
    """Generate semantic embedding for text."""
    model = _get_embedding_model()
    if model is None:
        # Fallback to hash-based (for backward compatibility)
        import hashlib
        h = hashlib.sha256(text.encode()).digest()
        return [float(b) / 255.0 for b in h] + [0.0] * (384 - len(h))

    embedding = model.encode(text, normalize_embeddings=True)
    return embedding.tolist()
```

```python
def _compress_tool_result(result_str: str, max_chars: int = 8000) -> str:
    """Compress tool result intelligently."""
    if len(result_str) <= max_chars:
        return result_str

    # Strategy 1: Try to find natural break points (JSON objects, sections)
    # Look for the last complete JSON object or section before the limit
    truncated = result_str[:max_chars]

    # Try to find last complete JSON object
    last_brace = truncated.rfind("}")
    if last_brace > max_chars * 0.7:  # If we found a brace in the last 30%
        truncated = truncated[:last_brace + 1]

    # Strategy 2: If still too large, try to summarize
    if len(truncated) > max_chars:
        # Take first 4000 + last 2000 + summary marker
        head = result_str[:4000]
        tail = result_str[-2000:]
        truncated = f"{head}\n\n... [truncated {len(result_str) - 6000} chars] ...\n\n{tail}"

    return truncated
```

```python
import logging
from datetime import datetime, timedelta, timezone

    severity: str  # "warning", "critical"
    recommendation: str

class RegressionDetector:
    """Monitors performance metrics for degradation."""

    def __init__(
        self,
        min_trades_for_alert: int = 20,
        win_rate_warning: float = 0.40,
        win_rate_critical: float = 0.30,
        drawdown_warning: float = 0.10,
        drawdown_critical: float = 0.20,
        calibration_warning: float = 0.60,

        self.win_rate_warning = win_rate_warning
        self.win_rate_critical = win_rate_critical
        self.drawdown_warning = drawdown_warning
        self.drawdown_critical = drawdown_critical
        self.calibration_warning = calibration_warning

    async def check_regression(self) -> list[RegressionAlert]:
        """Check all metrics for regression. Returns list of alerts."""
        alerts = []
        alerts.extend(await self._check_win_rate())
        alerts.extend(await self._check_drawdown())
        alerts.extend(await self._check_calibration())
        alerts.extend(await self._check_loss_streak())
        return alerts
```

```python
"""A/B testing framework for trading strategies."""
import logging
import hashlib
import random
from datetime import datetime, timezone
from typing import Optional
from dataclasses import dataclass, field

logger = logging.getLogger(__name__)

@dataclass
class Experiment:
    """An A/B test experiment."""
    experiment_id: str
    name: str
    variant_a: str  # Control
    variant_b: str  # Treatment
    start_date: str = ""
    end_date: str = ""
    sample_size_a: int = 0
    sample_size_b: int = 0
    conversion_a: float = 0.0  # Win rate for variant A
    conversion_b: float = 0.0  # Win rate for variant B
    is_active: bool = True

@dataclass
class ExperimentResult:
    """Statistical result of an A/B test."""
    experiment_id: str
    winner: str  # "A", "B", or "inconclusive"
    confidence: float  # 0-1
    lift: float  # % improvement of winner over loser
    sample_size: int
    recommendation: str

class ABTestingFramework:
    """Manages A/B testing for trading strategies."""

    def __init__(self):
        self._experiments: dict[str, Experiment] = {}

    def create_experiment(
        self,
        name: str,
        variant_a: str,
        variant_b: str,
        experiment_id: Optional[str] = None,
    ) -> Experiment:
        """Create a new A/B test experiment."""
        if experiment_id is None:
            experiment_id = hashlib.md5(f"{name}{datetime.now()}".encode()).hexdigest()[:8]

        exp = Experiment(
            experiment_id=experiment_id,
            name=name,
            variant_a=variant_a,
            variant_b=variant_b,
            start_date=datetime.now(timezone.utc).isoformat(),
        )
        self._experiments[experiment_id] = exp
        logger.info("Created experiment %s: %s vs %s", experiment_id, variant_a, variant_b)
        return exp

    def assign_variant(self, experiment_id: str, subject: str) -> str:
        """Deterministically assign a subject to variant A or B."""
        exp = self._experiments.get(experiment_id)
        if not exp:
            return "A"

        # Hash-based assignment (consistent for same subject)
        h = hashlib.md5(f"{experiment_id}:{subject}".encode()).hexdigest()
        return "A" if int(h, 16) % 2 == 0 else "B"

    def record_outcome(
        self,
        experiment_id: str,
        variant: str,
        subject: str,
        outcome: float,  # P&L percentage
    ):
        """Record the outcome for a variant assignment."""
        exp = self._experiments.get(experiment_id)
        if not exp:
            return

        if variant == "A":
            exp.sample_size_a += 1
        else:
            exp.sample_size_b += 1

    def compute_results(self, experiment_id: str) -> Optional[ExperimentResult]:
        """Compute statistical results for an experiment."""
        exp = self._experiments.get(experiment_id)
        if not exp:
            return None

        total = exp.sample_size_a + exp.sample_size_b
        if total < 20:  # Minimum sample size
            return ExperimentResult(
                experiment_id=experiment_id,
                winner="inconclusive",
                confidence=0.0,
                lift=0.0,
                sample_size=total,
                recommendation=f"Need at least 20 samples, have {total}",
            )

        # Simple comparison (could use scipy for proper statistical test)
        if exp.conversion_a > exp.conversion_b:
            winner = "A"
            lift = ((exp.conversion_a - exp.conversion_b) / max(exp.conversion_b, 0.01)) * 100
        elif exp.conversion_b > exp.conversion_a:
            winner = "B"
            lift = ((exp.conversion_b - exp.conversion_a) / max(exp.conversion_a, 0.01)) * 100
        else:
            winner = "inconclusive"
            lift = 0.0

        # Confidence based on sample size (simplified)
        confidence = min(0.95, 0.5 + (total / 200))

        return ExperimentResult(
            experiment_id=experiment_id,
            winner=winner,
            confidence=confidence,
            lift=lift,
            sample_size=total,
            recommendation=f"Variant {winner} wins with {confidence:.0%} confidence" if winner != "inconclusive" else "Insufficient data",
        )

    def list_experiments(self) -> list[Experiment]:
        """List all experiments."""
        return list(self._experiments.values())
```
